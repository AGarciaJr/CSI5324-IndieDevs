		\end{call}

{\bfseries Admin - Review Report}
\bigskip
\begin{sequencediagram}
	
	\newthread{a}{Admin}
	\newinst[1]{c}{ReportController}
	\newinst[1]{s}{ReportService}
	\newinst[1]{r}{ReportRepository}
	
	\begin{call}{a}{reviewReport(reportId, decision)}{c}{success}
		
		\begin{call}{c}{getReport(reportId)}{s}{report}
			
			\begin{call}{s}{findReportById(reportId)}{r}{report}
			\end{call}
			
		\end{call}
		
		\begin{call}{c}{resolveReport(report, decision)}{s}{resolvedReport}
			
			\begin{call}{s}{[if decision.removeContent] hideContent(report.target)}{s}{}
			\end{call}
			
			\begin{call}{s}{[if decision.banUser] banUser(report.target.poster)}{s}{}
			\end{call}
			
			\begin{call}{s}{save(report)}{r}{updatedReport}
			\end{call}
			
		\end{call}
		
	\end{call}
	
\end{sequencediagram}
